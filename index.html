import { initializeApp, cert, getApps } from 'firebase-admin/app'; import { getAuth } from 'firebase-admin/auth'; import { getFirestore, FieldValue } from 'firebase-admin/firestore'; const json = (body, status = 200) => new Response(JSON.stringify(body), { status, headers: { 'content-type': 'application/json', 'cache-control': 'no-store' } }); const appFor = (env) => getApps()[0] || initializeApp({ credential: cert(JSON.parse(env.FIREBASE_SERVICE_ACCOUNT_JSON)) }); async function account(request, env) { const header = request.headers.get('authorization') || ''; if (!header.startsWith('Bearer ')) throw new Error('Sign in is required.'); return getAuth(appFor(env)).verifyIdToken(header.slice(7)); } async function member(request, env) { const token = await account(request, env), db = getFirestore(appFor(env)); const profile = (await db.doc(`users/${token.uid}`).get()).data(); if (!profile?.businessId) throw new Error('Your business profile is unavailable.'); return { uid: token.uid, profile, db }; } async function paystack(path, env, options = {}) { const response = await fetch(`https://api.paystack.co${path}`, { ...options, headers: { Authorization: `Bearer ${env.PAYSTACK_SECRET_KEY}`, 'content-type': 'application/json', ...(options.headers || {}) } }); const result = await response.json(); if (!response.ok || !result.status) throw new Error(result.message || 'Paystack request failed.'); return result.data; } function subscriptionRef(db, businessId) { return db.doc(`businesses/${businessId}/subscriptions/current`); } async function ensureTrial({ uid, profile, db }) { const ref = subscriptionRef(db, profile.businessId), existing = await ref.get(); if (existing.exists) return existing.data(); // Only a newly-created profile carries this immutable eligibility flag. if (!profile.trialEligible) return { status: 'trial_expired', plan: 'free' }; // Firebase's server timestamp is the source of truth; browser/device time is never used. const start = profile.createdAt?.toDate?.() || new Date(); const end = new Date(start); end.setUTCDate(end.getUTCDate() + 60); const record = { plan: 'premium', status: 'free_trial', trialStart: start, trialEnd: end, createdAt: FieldValue.serverTimestamp(), updatedAt: FieldValue.serverTimestamp() }; await ref.set(record); await db.doc(`users/${uid}`).update({ trialInitialized: true, updatedAt: FieldValue.serverTimestamp() }); return record; } async function status(request, env) { const ctx = await member(request, env), record = await ensureTrial(ctx); const now = Date.now(), trialEnd = record.trialEnd?.toDate?.().getTime(); const state = record.status === 'free_trial' && trialEnd && trialEnd <= now ? 'trial_expired' : record.status; if (state !== record.status) await subscriptionRef(ctx.db, ctx.profile.businessId).update({ status: state, updatedAt: FieldValue.serverTimestamp() }); return json({ status: state, plan: record.plan, trialEndsAt: record.trialEnd?.toDate?.().toISOString() || null, renewalAt: record.renewalAt?.toDate?.().toISOString() || null }); } async function checkout(request, env) { const ctx = await member(request, env), record = await ensureTrial(ctx); if (record.status === 'free_trial' && record.trialEnd?.toDate?.() > new Date()) return json({ error: 'Your two-month Premium trial is already active.' }, 409); const reference = `kf_${ctx.uid.slice(0, 10)}_${Date.now()}`; const callback = `${env.APP_ORIGIN}/?payment=complete`; const data = await paystack('/transaction/initialize', env, { method: 'POST', body: JSON.stringify({ email: ctx.profile.email, amount: 250000, plan: env.PAYSTACK_PLAN_CODE_MONTHLY, reference, callback_url: callback, metadata: { uid: ctx.uid, businessId: ctx.profile.businessId, product: 'KudiFlow Premium' } }) }); await subscriptionRef(ctx.db, ctx.profile.businessId).set({ paymentReference: reference, updatedAt: FieldValue.serverTimestamp() }, { merge: true }); return json({ authorizationUrl: data.authorization_url }); } async function signedWebhook(request, env) { const raw = await request.text(), signature = request.headers.get('x-paystack-signature') || ''; const key = await crypto.subtle.importKey('raw', new TextEncoder().encode(env.PAYSTACK_SECRET_KEY), { name: 'HMAC', hash: 'SHA-512' }, false, ['sign']); const hash = Array.from(new Uint8Array(await crypto.subtle.sign('HMAC', key, new TextEncoder().encode(raw)))).map(x => x.toString(16).padStart(2, '0')).join(''); if (signature.length !== hash.length || !crypto.timingSafeEqual?.(new TextEncoder().encode(signature), new TextEncoder().encode(hash))) { if (signature !== hash) return new Response('Unauthorized', { status: 401 }); } const event = JSON.parse(raw), data = event.data || {}, meta = data.metadata || {}, businessId = meta.businessId; if (!businessId) return new Response('Ignored', { status: 200 }); const db = getFirestore(appFor(env)), ref = subscriptionRef(db, businessId); if (event.event === 'charge.success') { const verified = await paystack(`/transaction/verify/${encodeURIComponent(data.reference)}`, env); if (verified.status !== 'success') return new Response('Unverified', { status: 400 }); await ref.set({ plan: 'premium', status: 'active', paymentReference: data.reference, paystackCustomerCode: verified.customer?.customer_code || null, paystackSubscriptionCode: verified.subscription?.subscription_code || null, renewalAt: verified.paid_at ? new Date(new Date(verified.paid_at).getTime() + 30 * 86400000) : null, updatedAt: FieldValue.serverTimestamp() }, { merge: true }); } if (['subscription.disable', 'invoice.payment_failed'].includes(event.event)) await ref.set({ status: 'subscription_expired', updatedAt: FieldValue.serverTimestamp() }, { merge: true }); return new Response('OK'); } export default { async fetch(request, env) { try { const url = new URL(request.url); if (request.method === 'GET' && url.pathname === '/api/subscription/status') return status(request, env); if (request.method === 'POST' && url.pathname === '/api/subscription/checkout') return checkout(request, env); if (request.method === 'POST' && url.pathname === '/api/paystack/webhook') return signedWebhook(request, env); return new Response('Not found', { status: 404 }); } catch (error) { return json({ error: error.message || 'Request failed.' }, error.message === 'Sign in is required.' ? 401 : 400); } } };
